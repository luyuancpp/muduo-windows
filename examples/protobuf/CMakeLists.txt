include_directories(${PROJECT_SOURCE_DIR}/contrib/windows/include)

if (TARGET protobuf::libprotobuf)
  set(MUDUO_PROTOBUF_LIB protobuf::libprotobuf)
elseif(TARGET libprotobuf)
  set(MUDUO_PROTOBUF_LIB libprotobuf)
elseif(TARGET protobuf)
  set(MUDUO_PROTOBUF_LIB protobuf)
else()
  message(FATAL_ERROR "No protobuf library target found")
endif()

if (TARGET protobuf::libprotobuf-lite)
  set(MUDUO_PROTOBUF_LITE_LIB protobuf::libprotobuf-lite)
elseif(TARGET libprotobuf-lite)
  set(MUDUO_PROTOBUF_LITE_LIB libprotobuf-lite)
elseif(TARGET protobuf-lite)
  set(MUDUO_PROTOBUF_LITE_LIB protobuf-lite)
else()
  # Fall back to full protobuf when lite target is unavailable.
  set(MUDUO_PROTOBUF_LITE_LIB ${MUDUO_PROTOBUF_LIB})
endif()

if (TARGET zlibstatic)
  set(MUDUO_ZLIB_LIB zlibstatic)
elseif(TARGET zlib)
  set(MUDUO_ZLIB_LIB zlib)
elseif(TARGET ZLIB::ZLIB)
  set(MUDUO_ZLIB_LIB ZLIB::ZLIB)
else()
  message(FATAL_ERROR "No zlib library target found")
endif()

add_subdirectory(codec)
add_subdirectory(rpc)
add_subdirectory(rpcbalancer)
add_subdirectory(rpcbench)

if(CARES_INCLUDE_DIR AND CARES_LIBRARY)
  add_subdirectory(resolver)
else()
  add_subdirectory(resolver EXCLUDE_FROM_ALL)
endif()
